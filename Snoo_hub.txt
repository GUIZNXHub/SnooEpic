local function MoveToPoint(Target)
	if not Root then return true end

	local Difference = Target - Root.Position
	local Distance = Difference.Magnitude

	if Distance <= 1 then
		return true
	end

	local DeltaTime = RunService.Heartbeat:Wait()
	local Speed = math.max(TravelValue, 1)
	local Amount = math.min(Distance, Speed * DeltaTime)

	Root.CFrame = CFrame.new(
		Root.Position + Difference.Unit * Amount
	)

	return false
end

local function StartTravel()
	if Traveling then
		return
	end

	if not SavedPosition then
		TravelButton.Text = "❌  SALVE UMA LOCALIZAÇÃO"

		task.delay(1.5, function()
			if TravelButton.Parent then
				TravelButton.Text = "🚀  IR FLUTUANDO ATÉ LOCAL"
			end
		end)

		return
	end

	if not Root or not Humanoid then
		return
	end

	local Number = tonumber(TravelBox.Text)
	if Number then
		TravelValue = math.clamp(Number, 1, 1000)
	end

	Traveling = true
	CancelTravel = false
	TravelButton.Text = "✖  CANCELAR TRAJETO"

	local Destination = SavedPosition.Position

	-- Altura segura acima do destino.
	local FlightHeight = math.max(
		Root.Position.Y + 25,
		Destination.Y + 25
	)

	local OldAutoRotate = Humanoid.AutoRotate
	local OldPlatformStand = Humanoid.PlatformStand

	Humanoid.AutoRotate = false
	Humanoid.PlatformStand = true

	-- Impede a gravidade de derrubar o personagem durante o trajeto.
	Root.AssemblyLinearVelocity = Vector3.zero
	Root.AssemblyAngularVelocity = Vector3.zero

	local function MoveFloating(Target)
		if not Root then
			return true
		end

		local Difference = Target - Root.Position
		local Distance = Difference.Magnitude

		if Distance <= 0.75 then
			return true
		end

		local DeltaTime = RunService.Heartbeat:Wait()
		local Amount = math.min(
			Distance,
			math.max(TravelValue, 1) * DeltaTime
		)

		Root.CFrame = CFrame.new(
			Root.Position + Difference.Unit * Amount
		)

		Root.AssemblyLinearVelocity = Vector3.zero
		Root.AssemblyAngularVelocity = Vector3.zero

		return false
	end

	-- 1. Sobe para a altura de voo.
	while Traveling and not CancelTravel do
		if MoveFloating(Vector3.new(
			Root.Position.X,
			FlightHeight,
			Root.Position.Z
		)) then
			break
		end
	end

	-- 2. Vai pelo alto até ficar exatamente em cima do destino.
	while Traveling and not CancelTravel do
		if MoveFloating(Vector3.new(
			Destination.X,
			FlightHeight,
			Destination.Z
		)) then
			break
		end
	end

	-- 3. Desce verticalmente até a altura salva.
	while Traveling and not CancelTravel do
		if MoveFloating(Destination) then
			break
		end
	end

	if Traveling and not CancelTravel then
		-- Fixa exatamente no ponto salvo antes de soltar.
		Root.CFrame = SavedPosition
		Root.AssemblyLinearVelocity = Vector3.zero
		Root.AssemblyAngularVelocity = Vector3.zero

		-- Agora o personagem é solto e a gravidade volta normalmente.
		Humanoid.PlatformStand = OldPlatformStand
		Humanoid.AutoRotate = OldAutoRotate

		TravelButton.Text = "✓  CHEGOU"

		task.delay(1.5, function()
			if TravelButton.Parent then
				TravelButton.Text = "🚀  IR FLUTUANDO ATÉ LOCAL"
			end
		end)
	else
		-- Cancelou: solta no lugar atual sem teleportar.
		Humanoid.PlatformStand = OldPlatformStand
		Humanoid.AutoRotate = OldAutoRotate
		Root.AssemblyLinearVelocity = Vector3.zero
		Root.AssemblyAngularVelocity = Vector3.zero

		TravelButton.Text = "🚀  IR FLUTUANDO ATÉ LOCAL"
	end

	Traveling = false
	CancelTravel = false
end

TravelButton.MouseButton1Click:Connect(function()
	if Traveling then
		CancelTravel = true
	else
		task.spawn(StartTravel)
	end
end)

local function StartStepTeleport()
	if StepTeleporting then return end

	if not SavedPosition then
		TPButton.Text = "❌  SALVE UMA LOCALIZAÇÃO"

		task.delay(1.5, function()
			TPButton.Text = "✦  TP POR PASSOS • IR"
		end)

		return
	end

	if not Root then return end

	local Number = tonumber(TPBox.Text)
	if Number then
		TPValue = math.clamp(Number, 1, 200)
	end

	StepTeleporting = true
	CancelStepTeleport = false
	TPButton.Text = "✖  CANCELAR TP"

	local Destination = SavedPosition.Position

	while StepTeleporting and not CancelStepTeleport do
		local Difference = Destination - Root.Position
		local Distance = Difference.Magnitude

		if Distance <= 2 then
			break
		end

		local Step = math.min(2, Distance)

		Root.CFrame = CFrame.new(
			Root.Position + Difference.Unit * Step
		)

		Root.AssemblyLinearVelocity = Vector3.zero
		task.wait(1 / TPValue)
	end

	if StepTeleporting and not CancelStepTeleport then
		Root.CFrame = SavedPosition
		TPButton.Text = "✓  CHEGOU"

		task.delay(1.5, function()
			if TPButton.Parent then
				TPButton.Text = "✦  TP POR PASSOS • IR"
			end
		end)
	else
		TPButton.Text = "✦  TP POR PASSOS • IR"
	end

	StepTeleporting = false
	CancelStepTeleport = false
end

TPButton.MouseButton1Click:Connect(function()
	if StepTeleporting then
		CancelStepTeleport = true
	else
		task.spawn(StartStepTeleport)
	end
end)

local function ConfigurePrompt(Prompt)
	if not Prompt:IsA("ProximityPrompt") then return end

	if InstantPromptEnabled then
		if Prompt:GetAttribute("SnooOriginalHold") == nil then
			Prompt:SetAttribute(
				"SnooOriginalHold",
				Prompt.HoldDuration
			)
		end

		Prompt.HoldDuration = 0
		Prompt.RequiresLineOfSight = false
	else
		local Original =
			Prompt:GetAttribute("SnooOriginalHold")

		if typeof(Original) == "number" then
			Prompt.HoldDuration = Original
		end
	end
end

local function UpdatePrompts()
	for _, Object in ipairs(workspace:GetDescendants()) do
		if Object:IsA("ProximityPrompt") then
			ConfigurePrompt(Object)
		end
	end
end

RunService.Stepped:Connect(function()
	if not Character or not Humanoid or not Root then
		return
	end

	if WalkEnabled
		and not Traveling
		and not StepTeleporting then

		Humanoid.WalkSpeed = WalkValue
	end

	if JumpEnabled then
		Humanoid.UseJumpPower = true
		Humanoid.JumpPower = JumpValue
	end

	if NoclipEnabled then
		for _, Object in ipairs(Character:GetDescendants()) do
			if Object:IsA("BasePart") then
				Object.CanCollide = false
			end
		end
	end

	if AntiRagdollEnabled then
		local State = Humanoid:GetState()

		if State == Enum.HumanoidStateType.Ragdoll
			or State == Enum.HumanoidStateType.FallingDown
			or State == Enum.HumanoidStateType.Physics then

			Humanoid:ChangeState(
				Enum.HumanoidStateType.GettingUp
			)
		end
	end
end)

local Dragging = false
local DragStart = nil
local StartPosition = nil

Header.InputBegan:Connect(function(Input)
	if Input.UserInputType == Enum.UserInputType.MouseButton1
		or Input.UserInputType == Enum.UserInputType.Touch then

		Dragging = true
		DragStart = Input.Position
		StartPosition = Main.Position
	end
end)

UserInputService.InputChanged:Connect(function(Input)
	if not Dragging then return end

	if Input.UserInputType == Enum.UserInputType.MouseMovement
		or Input.UserInputType == Enum.UserInputType.Touch then

		local Delta = Input.Position - DragStart

		Main.Position = UDim2.new(
			StartPosition.X.Scale,
			StartPosition.X.Offset + Delta.X,
			StartPosition.Y.Scale,
			StartPosition.Y.Offset + Delta.Y
		)
	end
end)

UserInputService.InputEnded:Connect(function(Input)
	if Input.UserInputType == Enum.UserInputType.MouseButton1
		or Input.UserInputType == Enum.UserInputType.Touch then

		Dragging = false
	end
end)
